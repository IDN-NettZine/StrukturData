import java.util.Scanner;

// Class Node
class Pasien {
    String nama;
    String keluhan;
    int prioritas; // 1 = Kritis, 2 = Mendesak, 3 = Reguler
    Pasien next; // Pointer 'next'

    public Pasien(String nama, String keluhan, int prioritas) {
        this.nama = nama;
        this.keluhan = keluhan;
        this.prioritas = prioritas;
        this.next = null;
    }
}

// 2. Class Single Linked List Non-Circular
// Ubah sesuai dengan nim masing-masing (contoh: Klinik098)
public class Klinik<NIM> { 
    Pasien pasienPertama; // Pengganti 'head'
    String kodeKlinik;
    
    public Klinik<NIM>(String nim) {
        this.kodeKlinik = nim.length() >= 3 ? nim.substring(nim.length() - 3) : nim;
    }
    
    public boolean isEmpty(){
        return pasienPertama == null;
    }
    
    // Method bantu untuk pasien Kritis (Prioritas 1 di paling depan)(JANGAN DIUBAH)
    public void tambahDepan(String nama, String keluhan) {
        Pasien baru = new Pasien(nama, keluhan, 1);
        baru.next = pasienPertama;
        pasienPertama = baru;
        System.out.println("Berhasil: Pasien " + nama + " masuk di antrean terdepan (Prioritas 1).");
    }

    // Method bantu untuk pasien Reguler (Prioritas 3 di paling belakang)(JANGAN DIUBAH)
    public void tambahBelakang(String nama, String keluhan) {
        Pasien baru = new Pasien(nama, keluhan, 3);
        if (pasienPertama == null) {
            pasienPertama = baru;
        } else {
            Pasien bantu = pasienPertama;
            while (bantu.next != null) {
                bantu = bantu.next;
            }
            bantu.next = baru;
        }
        System.out.println("Berhasil: Pasien " + nama + " masuk di antrean belakang (Prioritas 3).");
    }
    
    public void hapusDepan() {
        // Kondisi 1: Antrean kosong
        if (pasienPertama == null) {
            System.out.println("GAGAL: Antrean kosong, tidak ada pasien untuk dipanggil.");
            return;
        }

        // Kondisi 2: Simpan referensi pasien terdepan, lalu geser pasienPertama ke node berikutnya
        Pasien dipanggil = pasienPertama;
        pasienPertama = pasienPertama.next;
        dipanggil.next = null; // Putuskan tautan node lama agar bersih dari rantai list

        System.out.println("PANGGILAN MASUK: Pasien " + dipanggil.nama 
                           + " (Prioritas " + dipanggil.prioritas + " | Keluhan: " + dipanggil.keluhan 
                           + ") silakan menuju ruang periksa.");
    }

    // Method menampilkan seluruh antrean (JANGAN DIUBAH)
    public void cetakAntrean() {
        System.out.println("\n--- DAFTAR ANTREAN KLINIK (" + kodeKlinik + ") ---");
        if (pasienPertama == null) {
            System.out.println("Antrean Kosong (NULL)");
            return;
        }
        Pasien bantu = pasienPertama;
        int nomerUrut = 1;
        while (bantu != null) {
            
            System.out.print("[" + kodeKlinik + "-"+ nomerUrut + " | Level" + bantu.prioritas + " | " 
                             + bantu.nama + " (" + bantu.keluhan + ")] -> ");
            bantu = bantu.next;
            nomerUrut++;
        }
        System.out.println("NULL");
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        System.out.println("=== SISTEM ANTREAN PRIORITAS KLINIK ===");
        System.out.print("Masukkan Nomor ID Pegawai Klinik: "); // Isi dengan NIM masing-masing saat dijalankan
        String nim = input.nextLine().trim();

        Klinik<NIM></NIM> klinik = new Klinik<NIM></NIM>(nim);
        
        klinik.tambahDepan("Anna", "Sesak");
        klinik.tambahDepan("Budi", "Pass-out");
        klinik.cetakAntrean();
        System.out.println("");
        klinik.tambahBelakang("Ruri", "Demam");
        klinik.tambahBelakang("Dede", "Flu");
        klinik.cetakAntrean();
        System.out.println("");
        klinik.hapusDepan();
